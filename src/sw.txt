// Service worker: omogućava instalaciju na mobitel i prikazuje igru i kad je internet slab.
// Stranica se uvijek prvo traži s mreže (da uvijek dobiješ najnoviju verziju), keš je samo rezerva.
const CACHE = 'aso-v1';
const SHELL = ['/', '/manifest.webmanifest', '/icon-192.png', '/apple-touch-icon.png'];

self.addEventListener('install', e => {
  e.waitUntil(caches.open(CACHE).then(c => c.addAll(SHELL)).then(() => self.skipWaiting()));
});

self.addEventListener('activate', e => {
  e.waitUntil(caches.keys()
    .then(keys => Promise.all(keys.filter(k => k !== CACHE).map(k => caches.delete(k))))
    .then(() => self.clients.claim()));
});

self.addEventListener('fetch', e => {
  const req = e.request;
  const url = new URL(req.url);
  if (req.method !== 'GET' || url.origin !== location.origin) return;
  if (url.pathname.startsWith('/api/') || url.pathname.startsWith('/ws/')) return;   // igra uvijek ide uživo
  if (req.mode === 'navigate') {
    e.respondWith(fetch(req).then(res => {
      const copy = res.clone();
      if (res.ok) caches.open(CACHE).then(c => c.put('/', copy));
      return res;
    }).catch(() => caches.match('/')));
    return;
  }
  if (SHELL.includes(url.pathname) || url.pathname.endsWith('.png')) {
    e.respondWith(caches.match(req).then(hit => hit || fetch(req)));
  }
});
